// โค้ด Google Apps Script สำหรับระบบตรวจบ้าน 162/307
function doGet(e) {
  var data = getOrCreateStore();
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var payload = JSON.parse(e.postData.contents);
    var store = getOrCreateStore();
    if (payload.action === 'sync') {
      store.items = payload.items || [];
      store.meta = payload.meta || store.meta;
      saveStore(store);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', data: store }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateStore() {
  var files = DriveApp.getFilesByName("defect_db_162_307.json");
  if (files.hasNext()) {
    var content = files.next().getBlob().getDataAsString();
    return JSON.parse(content);
  }
  var init = { meta: { projectName: "เดอะแพลนท์ นิวทาวน์ บางนา-เทพารักษ์", date: "18-10-2569" }, items: [] };
  DriveApp.createFile("defect_db_162_307.json", JSON.stringify(init), "application/json");
  return init;
}

function saveStore(data) {
  var files = DriveApp.getFilesByName("defect_db_162_307.json");
  if (files.hasNext()) {
    files.next().setContent(JSON.stringify(data));
  }
}
